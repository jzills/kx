{{- /* The released versions, newest first, and the latest of them. {{ $releases := partialCached "kx/releases.html" . }} {{ $releases.latest }} → "0.10.2" {{ range $releases.notes }} → dicts of version and body The latest version is pyproject.toml's, which release.yml stamps when a release publishes, not the highest file in release-notes/: the paragraph for a release is written and merged before the release branch is pushed, so for a while the newest file names a version nobody can install yet. Versions are compared as numbers, major/minor/patch packed into one integer. Sorted as strings, 0.9.2 would land above 0.10.0. */ -}} {{- $latest := "" -}} {{- with resources.Get "pyproject.toml" -}} {{- $latest = (. | transform.Unmarshal).project.version -}} {{- end -}} {{- $key := partial "kx/version-key.html" $latest -}} {{- $notes := slice -}} {{- range resources.Match "release-notes/v*.md" -}} {{- $version := strings.TrimPrefix "v" (path.BaseName .Name) -}} {{- $rank := partial "kx/version-key.html" $version -}} {{- if le $rank $key -}} {{- $notes = $notes | append (dict "version" $version "rank" $rank "body" .Content) -}} {{- end -}} {{- end -}} {{- return dict "latest" $latest "notes" (sort $notes "rank" "desc") -}}
